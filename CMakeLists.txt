cmake_minimum_required(VERSION 3.22)
project(TS3SoundBridge VERSION 1.1.1 LANGUAGES CXX)
configure_file(src/version.h.in generated/soundbridge_version.h @ONLY)
configure_file(src/windows/icon_path.h.in generated/icon_path.h @ONLY)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /permissive- /utf-8)
endif()

add_library(soundbridge_core INTERFACE)
target_include_directories(soundbridge_core INTERFACE src "${CMAKE_BINARY_DIR}/generated")
include(CTest)
if(BUILD_TESTING)
  add_executable(soundbridge_tests tests/audio_tests.cpp)
  target_link_libraries(soundbridge_tests PRIVATE soundbridge_core)
  add_test(NAME audio_core COMMAND soundbridge_tests)
endif()

if(WIN32)
  enable_language(RC)
  add_executable(soundbridge_icon_tool tools/render_icon.cpp)
  target_compile_definitions(soundbridge_icon_tool PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(soundbridge_icon_tool PRIVATE d2d1 d3d11 dxgi windowscodecs ole32 shlwapi)
  add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/icons/soundbridge.ico" "${CMAKE_BINARY_DIR}/icons/icon-32.png" "${CMAKE_BINARY_DIR}/icons/icon-256.png"
    COMMAND soundbridge_icon_tool "${CMAKE_SOURCE_DIR}/assets/icon.svg" "${CMAKE_BINARY_DIR}/icons"
    DEPENDS soundbridge_icon_tool assets/icon.svg VERBATIM)
  add_custom_target(soundbridge_icons DEPENDS "${CMAKE_BINARY_DIR}/icons/soundbridge.ico")
  set(TS3_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/ts3-sdk" CACHE PATH "Pinned TS3 SDK root (run scripts/fetch-sdk.ps1)")
  if(NOT EXISTS "${TS3_SDK_DIR}/include/ts3_functions.h")
    message(FATAL_ERROR "TS3 SDK missing. Run scripts/fetch-sdk.ps1 or set TS3_SDK_DIR.")
  endif()
  add_library(soundbridge_capture STATIC src/windows/loopback_capture.cpp)
  target_compile_definitions(soundbridge_capture PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_link_libraries(soundbridge_capture PUBLIC soundbridge_core ole32 mmdevapi avrt)
  add_library(soundbridge_engine STATIC src/settings.cpp src/bridge_engine.cpp src/windows/applications.cpp)
  target_link_libraries(soundbridge_engine PUBLIC soundbridge_capture)
  set(SOUNDBRIDGE_JSON_DIR "${CMAKE_SOURCE_DIR}/.deps/json" CACHE PATH "Pinned nlohmann JSON headers")
  if(NOT EXISTS "${SOUNDBRIDGE_JSON_DIR}/nlohmann/json.hpp")
    message(FATAL_ERROR "JSON dependency missing. Run scripts/fetch-dependencies.ps1.")
  endif()
  add_library(soundbridge_updates STATIC src/update_checker.cpp)
  target_include_directories(soundbridge_updates SYSTEM PRIVATE "${SOUNDBRIDGE_JSON_DIR}")
  target_link_libraries(soundbridge_updates PUBLIC soundbridge_engine winhttp)
  add_library(ts3_soundbridge SHARED src/plugin.cpp src/windows/settings_window.cpp src/windows/settings_dialog.rc)
  target_include_directories(ts3_soundbridge PRIVATE "${TS3_SDK_DIR}/include" src/windows)
  target_link_libraries(ts3_soundbridge PRIVATE soundbridge_updates comctl32 uxtheme shell32)
  add_dependencies(ts3_soundbridge soundbridge_icons)
  set_target_properties(ts3_soundbridge PROPERTIES PREFIX "" OUTPUT_NAME "ts3_soundbridge_win64")
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This plugin supports x64 TS3 only. Configure with -A x64.")
  endif()
  add_executable(soundbridge_probe src/probe.cpp)
  target_link_libraries(soundbridge_probe PRIVATE soundbridge_engine winmm)
  if(BUILD_TESTING)
    add_executable(soundbridge_plugin_tests tests/plugin_tests.cpp)
    target_compile_definitions(soundbridge_plugin_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    target_include_directories(soundbridge_plugin_tests PRIVATE "${TS3_SDK_DIR}/include" src)
    target_link_libraries(soundbridge_plugin_tests PRIVATE comctl32 gdiplus)
    add_executable(soundbridge_selection_tests tests/selection_tests.cpp)
    target_link_libraries(soundbridge_selection_tests PRIVATE soundbridge_engine)
    add_test(NAME app_selection COMMAND soundbridge_selection_tests)
    add_executable(soundbridge_update_tests tests/update_tests.cpp)
    target_link_libraries(soundbridge_update_tests PRIVATE soundbridge_updates)
    add_test(NAME update_checker COMMAND soundbridge_update_tests)
    add_dependencies(soundbridge_plugin_tests ts3_soundbridge)
    add_test(NAME plugin_contract COMMAND soundbridge_plugin_tests $<TARGET_FILE:ts3_soundbridge>)
  endif()
endif()
